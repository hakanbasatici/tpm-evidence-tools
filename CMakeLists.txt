cmake_minimum_required(VERSION 3.20)
project(TpmEvidenceTools VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
if(BUILD_TESTING)
  add_executable(ProfileAndInputTests tests/ProfileAndInput.test.cpp)
  target_include_directories(ProfileAndInputTests PRIVATE src)
  if(MSVC)
    target_compile_options(ProfileAndInputTests PRIVATE /UNDEBUG)
  else()
    target_compile_options(ProfileAndInputTests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME profile-and-input COMMAND ProfileAndInputTests)
endif()
if(WIN32)
  enable_language(RC)
  add_executable(PcrBanks src/PcrBanks.cpp src/Version.rc)
  add_executable(EphemeralClaim src/EphemeralClaim.cpp src/Version.rc)
  target_link_libraries(PcrBanks PRIVATE tbs)
  target_link_libraries(EphemeralClaim PRIVATE ncrypt bcrypt tbs)
  foreach(target PcrBanks EphemeralClaim)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /EHsc)
    endif()
  endforeach()
endif()
